cmake_minimum_required(VERSION 3.16)
project(mini-redis-cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(mini-redis src/main.cpp src/kv.cpp)
target_include_directories(mini-redis PRIVATE include)
target_compile_options(mini-redis PRIVATE -Wall -Wextra -Wpedantic -Werror -D_FORTIFY_SOURCE=2 -fstack-protector-strong)
if(WIN32)
  target_link_libraries(mini-redis PRIVATE ws2_32)
else()
  target_link_libraries(mini-redis PRIVATE pthread)
endif()
